  }

% Newest first.
\DeclareSortingTemplate{datedesc}{
  \sort[direction=descending]{\field{year}}
  \sort[direction=descending]{\field{month}}
  \sort[direction=descending]{\field{day}}
  \sort{\field{title}}
}

\DeclareFieldFormat{usera}{\href{#1}{video}}
\DeclareFieldFormat{userb}{\href{#1}{slides}}
\DeclareFieldFormat[misc]{url}{\href{#1}{event}}
\DeclareFieldFormat[misc]{title}{\textbf{#1}}

% One talk: Title. Event, Host, Location. Note. [event | slides | video]
\DeclareBibliographyDriver{misc}{%
  \printfield{title}%
  \newunit\newblock
  \printfield{eventtitle}%
  \setunit{\addcomma\space}%
  \printlist{organization}%
  \setunit{\addcomma\space}%
  \printlist{location}%
  \newunit\newblock
  \printfield{note}%
  \newunit\newblock
  \ifboolexpr{test {\iffieldundef{url}} and test {\iffieldundef{userb}} and test {\iffieldundef{usera}}}
    {}
    {\printtext[brackets]{%
       \printfield{url}%
       \setunit{\addspace\textbar\addspace}%
       \printfield{userb}%
       \setunit{\addspace\textbar\addspace}%
       \printfield{usera}}}%
  \finentry}
